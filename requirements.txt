# ---- MySQL 驱动（二选一）----
# 1) 纯 Python 驱动（推荐 Windows，无需编译工具链）
#    实测说明：本机 conda 环境实际解析到 PyMySQL 1.4.6（conda-forge 包），
#    PyPI 上最新公开版本为 1.1.2；两者在本项目用到的接口上等价。
#    Django 的 "mysqlclient >= 1.4.3" 断言按 mysqlclient 版本体系制定，
#    与 PyMySQL 版本号不可直接比较，故由 smass/__init__.py 的适配 shim 兜底
#    （仅当 version_info < (1,4,3) 时才声明兼容版本，不覆盖真实 __version__）。
PyMySQL>=1.1.2,<2

# MySQL 8 服务端默认使用 caching_sha2_password；PyMySQL 在**没有** cryptography
# 时只能完成 fast-auth（缓存命中），一旦认证缓存被清空（例如 FLUSH PRIVILEGES）
# 就会抛 "cryptography package is required"。R2.2 G5 实测踩中该坑。
# 因此把 cryptography 列为**运行依赖**（等价于 PyMySQL[rsa]），用范围约束而非
# 锁死单机版本，避免在全平台上强制某个具体小版本。
cryptography>=41,<51

# 2) 或安装 mysqlclient（需要 MSVC 构建工具；装好后可完全去掉上述 shim）
# mysqlclient==2.2.7
